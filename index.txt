  <script>
    const audio = document.getElementById('bgMusic');
    const musicToggle = document.getElementById('musicToggle');
    const playBtnText = document.getElementById('playBtnText');
    const musicStatusText = document.getElementById('musicStatusText');
    let isPlaying = false;

    function startMusic() {
      if (!isPlaying) {
        audio.play().then(() => {
          isPlaying = true;
          playBtnText.textContent = "Pauză";
          musicStatusText.textContent = "Melodia noastră cântă... 🤠🎺";
        }).catch(() => {});
      }
    }

    function toggleMusic() {
      if (isPlaying) {
        audio.pause();
        isPlaying = false;
        playBtnText.textContent = "Pornește";
        musicStatusText.textContent = "Muzică oprită. Apasă pentru a continua! 🎶";
      } else {
        audio.play().then(() => {
          isPlaying = true;
          playBtnText.textContent = "Pauză";
          musicStatusText.textContent = "Melodia noastră cântă... 🤠🎺";
        });
      }
    }

    musicToggle.addEventListener('click', toggleMusic);

    // Sayfaya herhangi ilk tıklamada otomatik başlatma desteği
    document.body.addEventListener('click', function autoPlayOnce() {
      startMusic();
      document.body.removeEventListener('click', autoPlayOnce);
    }, { once: true });

    // Çay & Pasta Tepkileri
    const toast = document.getElementById('toastMsg');
    let teaSips = 0;
    const teaQuotes = [
      "Ai luat o înghițitură de ceai verde cald... 🌿",
      "Ceaiul acesta îți aduce liniște și energie bună! ✨",
      "Un zâmbet cald pentru cea mai scumpă fată! 🍵",
      "Să ai o zi la fel de senină ca sufletul tău! 💚"
    ];

    document.getElementById('teaBtn').addEventListener('click', () => {
      startMusic();
      teaSips++;
      confetti({ particleCount: 15, spread: 45, origin: { y: 0.65 } });
      toast.textContent = teaQuotes[teaSips % teaQuotes.length];
    });

    let cakeBites = 0;
    const cakeQuotes = [
      "O mușcătură delicioasă de tort cu multă ciocolată! 🍫",
      "Pentru că meriți cel mai dulce desert din lume! 🍰",
      "Dulceața asta e doar o zecime din cât ești tu de dulce! ❤️",
      "Ciocolată pentru fericire maximă azi! 🍓"
    ];

    document.getElementById('cakeBtn').addEventListener('click', () => {
      startMusic();
      cakeBites++;
      confetti({ particleCount: 20, spread: 55, origin: { y: 0.65 } });
      toast.textContent = cakeQuotes[cakeBites % cakeQuotes.length];
    });

    // Bilet Damgalama
    function claimTicket(el) {
      startMusic();
      if (!el.classList.contains('claimed')) {
        el.classList.add('claimed');
        el.querySelector('.ticket-badge').textContent = 'Validat! ✔️';
        confetti({ particleCount: 25, spread: 60, origin: { y: 0.7 } });
        toast.textContent = "Cupon activat cu succes! 🥰";
      }
    }

    // Affetme Butonu
    const forgiveBtn = document.getElementById('forgiveBtn');
    const modal = document.getElementById('modalOverlay');

    forgiveBtn.addEventListener('click', () => {
      startMusic();
      confetti({
        particleCount: 120,
        spread: 90,
        origin: { y: 0.55 },
        colors: ['#5a7d65', '#d8695b', '#f2a63b', '#caa573']
      });
      modal.style.display = 'flex';
    });

    function closeModal() {
      modal.style.display = 'none';
      toast.textContent = "Te iubesc din tot sufletul! ❤️";
    }
  </script>
